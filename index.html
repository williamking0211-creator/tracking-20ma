const express = require('express'); const cors = require('cors'); const YahooFinance = require('yahoo-finance2').default; const yahooFinance = new YahooFinance({ suppressNotices: ['yahooSurvey', 'ripHistorical'], validation: { logErrors: false } }); const app = express(); const PORT = 3000; app.use(cors()); app.use((req, res, next) => { res.header('Access-Control-Allow-Origin', '*'); res.header('Access-Control-Allow-Methods', 'GET, POST, OPTIONS'); function formatTwPrice(price) { if (typeof price !== 'number' || isNaN(price)) return 0; const clean = Math.round(price * 10000) / 10000; if (clean < 10) return Number(clean.toFixed(2)); if (clean < 50) return Number((Math.round(clean * 20) / 20).toFixed(2)); if (clean < 100) return Number(clean.toFixed(1)); if (clean < 500) return Number((Math.round(clean * 2) / 2).toFixed(1)); if (clean < 1000) return Math.round(clean); return Math.round(clean / 5) * 5; } let stockListCache = []; let latestCombinedStocks = []; let lastHistoryDate = ''; let isHistoryUpdating = false; let isRealtimeUpdating = false; async function getAllTaiwanStocks() { const [twseRes, tpexRes] = await Promise.all([ fetch('https://openapi.twse.com.tw/v1/exchangeReport/STOCK_DAY_ALL'), fetch('https://www.tpex.org.tw/openapi/v1/tpex_mainboard_daily_close_quotes') ]); const twseList = await twseRes.json(); const tpexList = await tpexRes.json(); const twseStocks = twseList .filter(item => /^[1-9]\d{3}$/.test(item.Code)) .map(item => ({ code: item.Code, name: (item.CompanyNa_me || item.CompanyNames || item.Name || '').trim(), suffix: '.TW', market: 'tse' })); const tpexStocks = tpexList .filter(item => /^[1-9]\d{3}$/.test(item.SecuritiesCompanyCode)) .map(item => ({ code: item.SecuritiesCompanyCode, name: (item.CompanyName || '').trim(), suffix: '.TWO', market: 'otc' })); return [...twseStocks, ...tpexStocks]; } // 獲取歷史資料並組合成每日 OHLC 陣列 async function fetchSingleHistory(stock) { const symbol = `${stock.code}${stock.suffix}`; const period2 = new Date(); const period1 = new Date(); period1.setDate(period2.getDate() - 45); try { const queryOptions = { period1: period1.toISOString().split('T')[0], period2: period2.toISOString().split('T')[0], interval: '1d' }; const chartData = await yahooFinance.chart(symbol, queryOptions); const quotes = chartData.quotes ? chartData.quotes.filter(q => q.close !== null && q.open !== null && q.high !== null && q.low !== null) : []; if (quotes.length < 21) return null; const todayTaipei = new Date().toLocaleDateString('en-CA', { timeZone: 'Asia/Taipei' }); const allBars = quotes.map(q => ({ date: new Date(q.date).toLocaleDateString('en-CA', { timeZone: 'Asia/Taipei' }), open: formatTwPrice(q.open), high: formatTwPrice(q.high), low: formatTwPrice(q.low), close: formatTwPrice(q.close), vol: Math.round((q.volume || 0) / 1000) // Yahoo Finance 單位為股，統一除以 1000 換算為「張」 })); // 區分「昨日(含以前)歷史線」與「今日線」 const lastBar = allBars[allBars.length - 1]; const hasTodayInChart = lastBar.date === todayTaipei; const history = hasTodayInChart ? allBars.slice(0, -1) : allBars; const todayFromChart = hasTodayInChart ? lastBar : null; if (history.length < 21) return null; const yestBar = history[history.length - 1]; const prevBar = history[history.length - 2]; const closes = history.map(h => h.close); const vols = history.map(h => h.vol); const prev20Closes = closes.slice(-21, -1); const yest20Closes = closes.slice(-20); const prev5Vols = vols.slice(-6, -1); const avgVol5 = prev5Vols.length > 0 ? calcMA(prev5Vols) : 1; const prevMa20 = Number(calcMA(prev20Closes).toFixed(2)); const yestMa20 = Number(calcMA(yest20Closes).toFixed(2)); return { code: stock.code, name: stock.name, market: stock.market, prevClose: prevBar.close, prevMa20, yestMa20, yestOpen: yestBar.open, yestHigh: yestBar.high, yestLow: yestBar.low, yestClose: yestBar.close, yestVol: yestBar.vol, avgVol5: Number(avgVol5.toFixed(0)), todayOpen: todayFromChart ? todayFromChart.open : 0, todayHigh: todayFromChart ? todayFromChart.high : 0, todayLow: todayFromChart ? todayFromChart.low : 0, todayClose: todayFromChart ? todayFromChart.close : 0, todayVol: todayFromChart ? todayFromChart.vol : 0, history }; } catch (err) { return null; } } async function initAndScheduleBackgroundSync() { const todayTaipei = new Date().toLocaleDateString('en-CA', { timeZone: 'Asia/Taipei' }); if (lastHistoryDate === todayTaipei) return; if (isHistoryUpdating) return; isHistoryUpdating = true; try { console.log('正在初始化上市櫃清單與歷史數據清算...'); stockListCache = await getAllTaiwanStocks(); let historyResults = []; const batchSize = 30; for (let i = 0; i < stockListCache.length; i += batchSize) { const batch = stockListCache.slice(i, i + batchSize); const promises = batch.map(s => fetchSingleHistory(s)); const results = await Promise.all(promises); historyResults.push(...results.filter(r => r !== null)); process.stdout.write(`\r歷史基準檔建立進度: ${Math.min(i + batchSize, stockListCache.length)} / ${stockListCache.length}`); } console.log(`\n基礎檔完成! 共 ${historyResults.length} 檔。`); latestCombinedStocks = historyResults; lastHistoryDate = todayTaipei; // 基準建立完成後立刻抓一次個股今日真實開高低收與成交量 await refreshRealtimePricesOnly(); } catch (err) { console.error('背景初始化失敗:', err); } finally { isHistoryUpdating = false; } } // 針對符合關鍵K候選池的個股，向證交所/櫃買中心抓取真實「今日開盤、最高、最低、收盤與當日成交量」 async function refreshRealtimePricesOnly() { if (isRealtimeUpdating || latestCombinedStocks.length === 0) return; isRealtimeUpdating = true; try { // 篩選出符合基本關鍵K條件的標的優先同步即時報價，加快速度並避免被證交所限流 const candidates = latestCombinedStocks.filter(s => { const gainPct = s.prevClose > 0 ? ((s.yestClose - s.prevClose) / s.prevClose) * 100 : 0; return s.prevClose <= s.prevMa20 && s.yestClose > s.yestMa20 && s.yestClose > s.yestOpen && gainPct >= 5.5; }); const targetList = candidates.length > 0 ? candidates : latestCombinedStocks.slice(0, 100); const realtimeMap = new Map(); const chunkSize = 40; for (let i = 0; i < targetList.length; i += chunkSize) { const chunk = targetList.slice(i, i + chunkSize); const exCh = chunk.map(s => `${s.market}_${s.code}.tw`).join('|'); const url = `https://mis.twse.com.tw/stock/api/getStockInfo.jsp?ex_ch=${encodeURIComponent(exCh)}&json=1&delay=0`; const res = await fetch(url, { headers: { 'User-Agent': 'Mozilla/5.0', 'Referer': 'https://mis.twse.com.tw/stock/fibest.jsp' } }).catch(() => null); if (res && res.ok) { const data = await res.json(); if (Array.isArray(data.msgArray)) { data.msgArray.forEach(item => { const code = item.c; const lastPrice = parseFloat(item.z) || parseFloat(item.y) || 0; const openPrice = parseFloat(item.o) || 0; const highPrice = parseFloat(item.h) || 0; const lowPrice = parseFloat(item.l) || 0; const totalVol = parseInt(item.v, 10) || 0; // MIS API 回傳的 v 單位即為「張」 if (code && lastPrice > 0) { realtimeMap.set(code, { open: openPrice > 0 ? formatTwPrice(openPrice) : formatTwPrice(lastPrice), high: highPrice > 0 ? formatTwPrice(highPrice) : formatTwPrice(lastPrice), low: lowPrice > 0 ? formatTwPrice(lowPrice) : formatTwPrice(lastPrice), price: formatTwPrice(lastPrice), vol: totalVol }); } }); } } } latestCombinedStocks.forEach(stock => { const rt = realtimeMap.get(stock.code); if (rt) { stock.todayOpen = rt.open; stock.todayHigh = Math.max(rt.high, rt.open, rt.price); stock.todayLow = Math.min(rt.low, rt.open, rt.price); stock.todayClose = rt.price; stock.price = rt.price; if (rt.vol > 0) stock.todayVol = rt.vol; } else { stock.todayOpen = stock.todayOpen || stock.yestClose; stock.todayHigh = stock.todayHigh || stock.yestClose; stock.todayLow = stock.todayLow || stock.yestClose; stock.todayClose = stock.todayClose || stock.yestClose; stock.price = stock.todayClose; } }); console.log(`[${new Date().toLocaleTimeString()}] 個股今日開高低收與成交量已更新 (同步 ${realtimeMap.size} 檔)`); } catch (err) { console.error('即時報價更新失敗:', err); } finally { isRealtimeUpdating = false; } } app.get('/stocks', (req, res) => { if (latestCombinedStocks.length === 0) { return res.status(503).json({ error: '後端剛啟動，正在載入歷史資料與計算基準，請稍候幾秒後重整網頁。' }); } res.json(latestCombinedStocks); }); app.listen(PORT, async () => { console.log(`Server running at http://localhost:${PORT}`); await initAndScheduleBackgroundSync(); setInterval(initAndScheduleBackgroundSync, 3600000); setInterval(refreshRealtimePricesOnly, 5000); });
